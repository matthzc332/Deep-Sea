using UnityEngine;

public class Gaviota_Picada : State_Base
{
    private Gaviota gaviota;

    void Awake()
    {
        gaviota = controlledObject.GetComponent<Gaviota>();
    }

    public override void EnterState()
    {
        Debug.Log("Gaviota entra en Picada (X + Y hacia barco).");
    }

    public override void UpdateState()
    {
        if (gaviota == null || gaviota.barco == null) return;

        // Dirección hacia el barco en X y Y
        Vector3 direccion = (gaviota.barco.position - controlledObject.transform.position).normalized;
        Vector3 movimiento = new Vector3(direccion.x, direccion.y, 0) * gaviota.velocidadPicada * Time.deltaTime;
        controlledObject.transform.Translate(movimiento);

        // Transición: vida <= 0 → Explotando
        if (gaviota.vida <= 0)
        {
            ExitState("Gaviota_Explotando");
            state_machine.SetState<Gaviota_Explotando>();
        }

        // Transición: colisión con barco → Explotando
        float dx = gaviota.barco.position.x - controlledObject.transform.position.x;
        float dy = gaviota.barco.position.y - controlledObject.transform.position.y;
        float distancia = Mathf.Sqrt(dx * dx + dy * dy);

        if (distancia < 1f) // ajustar según tamaño del barco
        {
            ExitState("Gaviota_Explotando");
            state_machine.SetState<Gaviota_Explotando>();
        }
    }

    public override void ExitState(string nextState)
    {
        Debug.Log($"Gaviota sale de Picada hacia {nextState}");
    }
}

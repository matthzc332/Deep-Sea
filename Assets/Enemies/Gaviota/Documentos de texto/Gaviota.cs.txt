using UnityEngine;

public class Gaviota : MonoBehaviour
{
    [Header("Velocidades")]
    public float velocidadNormal = 2f;   // Velocidad volando tranquilo (solo X)
    public float velocidadPicada = 5f;   // Velocidad en picada hacia el barco

    [Header("Límites de vuelo")]
    public float limiteXIzquierda = -5f; // Límite izquierdo de vuelo
    public float limiteXDerecha = 5f;    // Límite derecho de vuelo

    [Header("Vida y explosión")]
    public float vida = 10f;              // Vida de la gaviota
    public GameObject prefabExplosion;    // Prefab de la explosión
    public float duracionExplosion = 1f;  // Duración antes de destruirse

    [Header("Referencia al barco")]
    public Transform barco;               // Transform del barco objetivo

    [Header("Opciones adicionales")]
    public bool debug = true;             // Para activar logs

    void Update()
    {
        // Solo para test: reducir vida si presionas la tecla D
        if (Input.GetKeyDown(KeyCode.D))
        {
            vida -= 1f;
            if (debug) Debug.Log($"Gaviota recibe daño, vida actual: {vida}");
        }
    }
}

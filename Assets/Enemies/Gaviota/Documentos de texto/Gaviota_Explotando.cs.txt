using UnityEngine;

public class Gaviota_Explotando : State_Base
{
    private Gaviota gaviota;
    private bool explotando = false;

    void Awake()
    {
        gaviota = controlledObject.GetComponent<Gaviota>();
    }

    public override void EnterState()
    {
        if (gaviota == null) return;

        if (!explotando)
        {
            Debug.Log("Gaviota explota 💥");

            if (gaviota.prefabExplosion != null)
                GameObject.Instantiate(gaviota.prefabExplosion, controlledObject.transform.position, Quaternion.identity);

            explotando = true;
            GameObject.Destroy(controlledObject, gaviota.duracionExplosion);
        }
    }

    public override void UpdateState()
    {
        // No hace nada mientras explota
    }

    public override void ExitState(string nextState)
    {
        Debug.Log($"Gaviota sale de Explotando hacia {nextState}");
    }
}
